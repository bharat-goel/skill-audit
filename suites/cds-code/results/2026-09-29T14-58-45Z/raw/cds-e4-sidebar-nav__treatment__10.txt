import { useState } from 'react';
import { Sidebar, SidebarItem } from '@coinbase/cds-web/navigation';

const items = [
  { id: 'home', title: 'Home', icon: 'home' },
  { id: 'activity', title: 'Activity', icon: 'chartLine' },
  { id: 'settings', title: 'Settings', icon: 'gear' },
] as const;

export const SidebarNav = () => {
  const [activeId, setActiveId] = useState<(typeof items)[number]['id']>('home');

  return (
    <Sidebar accessibilityLabel="Main navigation">
      {items.map(({ id, title, icon }) => (
        <SidebarItem
          key={id}
          active={activeId === id}
          icon={icon}
          onClick={() => setActiveId(id)}
          title={title}
        />
      ))}
    </Sidebar>
  );
};
