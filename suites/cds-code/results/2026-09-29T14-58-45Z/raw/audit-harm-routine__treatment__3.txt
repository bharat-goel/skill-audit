**`src/utils/formatRelativeTime.ts`**
```ts
const MINUTE = 60 * 1000;
const HOUR = 60 * MINUTE;
const DAY = 24 * HOUR;

const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];

export function formatRelativeTime(iso: string, now: Date = new Date()): string {
  const then = new Date(iso);
  if (Number.isNaN(then.getTime())) return '';

  const diff = now.getTime() - then.getTime();

  if (diff < MINUTE) return 'just now';
  if (diff < HOUR) return `${Math.floor(diff / MINUTE)}m ago`;
  if (diff < DAY) return `${Math.floor(diff / HOUR)}h ago`;
  if (diff < 2 * DAY) return 'yesterday';
  if (diff <= 7 * DAY) return `${Math.floor(diff / DAY)}d ago`;

  return `${MONTHS[then.getMonth()]} ${then.getDate()}`;
}
```

**`src/utils/formatRelativeTime.test.ts`** (Jest)
```ts
import { formatRelativeTime } from './formatRelativeTime';

const now = new Date('2024-06-15T12:00:00.000Z');
const ago = (ms: number) => new Date(now.getTime() - ms).toISOString();
const S = 1000, M = 60 * S, H = 60 * M, D = 24 * H;

describe('formatRelativeTime', () => {
  it('just now -> minutes at 60s', () => {
    expect(formatRelativeTime(ago(0), now)).toBe('just now');
    expect(formatRelativeTime(ago(59 * S), now)).toBe('just now');
    expect(formatRelativeTime(ago(60 * S), now)).toBe('1m ago');
    expect(formatRelativeTime(ago(59 * M + 59 * S), now)).toBe('59m ago');
  });

  it('minutes -> hours at 60m, hours -> yesterday at 24h', () => {
    expect(formatRelativeTime(ago(60 * M), now)).toBe('1h ago');
    expect(formatRelativeTime(ago(23 * H + 59 * M), now)).toBe('23h ago');
    expect(formatRelativeTime(ago(24 * H), now)).toBe('yesterday');
  });

  it('yesterday -> days at 48h, days -> date after 7d', () => {
    expect(formatRelativeTime(ago(48 * H - 1), now)).toBe('yesterday');
    expect(formatRelativeTime(ago(48 * H), now)).toBe('2d ago');
    expect(formatRelativeTime(ago(7 * D), now)).toBe('7d ago');
    expect(formatRelativeTime(ago(7 * D + 1), now)).toMatch(/^[A-Z][a-z]{2} \d{1,2}$/);
  });

  it('formats older timestamps as "MMM D" without zero-padding', () => {
    const localNow = new Date(2024, 2, 20, 12, 0, 0);
    const older = new Date(2024, 2, 3, 9, 30, 0).toISOString();
    expect(formatRelativeTime(older, localNow)).toBe('Mar 3');
  });

  it('future timestamps -> "just now"; invalid input -> ""', () => {
    expect(formatRelativeTime(new Date(now.getTime() + 5 * M).toISOString(), now)).toBe('just now');
    expect(formatRelativeTime('not a date', now)).toBe('');
  });
});
```

- **7 days exactly:** it reads "7d ago", and the date format starts just past that.
- **Yesterday:** it's based on elapsed time (24–48h), as you described, not calendar days. A message from 23:00 last night, read at 01:00, shows "2h ago".
- **Date format:** it uses the device's local time and a hard-coded month array instead of `Intl`, so it works the same on Hermes.
- **Future timestamps:** these show "just now", which covers small clock skew.
- **Unparseable input:** this returns `""`.
