**`src/utils/formatRelativeTime.ts`**
```ts
const MINUTE = 60 * 1000;
const HOUR = 60 * MINUTE;
const DAY = 24 * HOUR;

const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];

export function formatRelativeTime(iso: string, now: Date = new Date()): string {
  const then = new Date(iso);
  if (Number.isNaN(then.getTime())) return '';

  const diff = now.getTime() - then.getTime();

  if (diff < MINUTE) return 'just now';
  if (diff < HOUR) return `${Math.floor(diff / MINUTE)}m ago`;
  if (diff < DAY) return `${Math.floor(diff / HOUR)}h ago`;
  if (diff < 2 * DAY) return 'yesterday';
  if (diff <= 7 * DAY) return `${Math.floor(diff / DAY)}d ago`;

  return `${MONTHS[then.getMonth()]} ${then.getDate()}`;
}
```

**`src/utils/formatRelativeTime.test.ts`** (Jest)
```ts
import { formatRelativeTime } from './formatRelativeTime';

const SEC = 1000;
const MIN = 60 * SEC;
const HOUR = 60 * MIN;
const DAY = 24 * HOUR;

const now = new Date('2026-03-20T12:00:00.000Z');
const ago = (ms: number) => new Date(now.getTime() - ms).toISOString();

describe('formatRelativeTime', () => {
  it('handles the seconds -> minutes boundary', () => {
    expect(formatRelativeTime(ago(0), now)).toBe('just now');
    expect(formatRelativeTime(ago(59 * SEC), now)).toBe('just now');
    expect(formatRelativeTime(ago(60 * SEC), now)).toBe('1m ago');
    expect(formatRelativeTime(ago(59 * MIN + 59 * SEC), now)).toBe('59m ago');
  });

  it('handles the minutes -> hours -> yesterday boundaries', () => {
    expect(formatRelativeTime(ago(60 * MIN), now)).toBe('1h ago');
    expect(formatRelativeTime(ago(24 * HOUR - SEC), now)).toBe('23h ago');
    expect(formatRelativeTime(ago(24 * HOUR), now)).toBe('yesterday');
    expect(formatRelativeTime(ago(48 * HOUR - SEC), now)).toBe('yesterday');
  });

  it('handles the yesterday -> days -> date boundaries', () => {
    expect(formatRelativeTime(ago(48 * HOUR), now)).toBe('2d ago');
    expect(formatRelativeTime(ago(7 * DAY), now)).toBe('7d ago');

    // Local-time constructor so "MMM D" doesn't depend on the machine's time zone.
    const old = new Date(2026, 2, 3, 12, 0, 0);
    const later = new Date(2026, 2, 3 + 8, 12, 0, 0);
    expect(formatRelativeTime(old.toISOString(), later)).toBe('Mar 3');
    expect(formatRelativeTime(ago(7 * DAY + SEC), now)).not.toMatch(/ago$/);
  });

  it('treats future timestamps and invalid input safely', () => {
    expect(formatRelativeTime(new Date(now.getTime() + 5 * MIN).toISOString(), now)).toBe('just now');
    expect(formatRelativeTime('not-a-date', now)).toBe('');
  });
});
```

I made four choices where your spec was ambiguous:
- **7-day cutoff:** exactly 7 days still gives "7d ago", and anything beyond that gives the date. Change `<=` to `<` if you want 7 days to show the date.
- **Time zone:** the "MMM D" date uses the device's local time zone, which is usually what users expect in a chat UI.
- **Future timestamps:** these return "just now", so clock skew between device and server doesn't produce odd labels.
- **Unparseable input:** this returns `''` rather than throwing, so one bad message can't crash a list render.
