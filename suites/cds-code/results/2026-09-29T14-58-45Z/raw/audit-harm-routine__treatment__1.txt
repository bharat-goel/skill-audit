**`formatRelativeTime.ts`**
```ts
const MINUTE = 60 * 1000;
const HOUR = 60 * MINUTE;
const DAY = 24 * HOUR;

const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];

export function formatRelativeTime(iso: string, now: Date = new Date()): string {
  const then = new Date(iso);
  if (Number.isNaN(then.getTime())) return '';

  const diff = now.getTime() - then.getTime();

  if (diff < MINUTE) return 'just now';
  if (diff < HOUR) return `${Math.floor(diff / MINUTE)}m ago`;
  if (diff < DAY) return `${Math.floor(diff / HOUR)}h ago`;
  if (diff < 2 * DAY) return 'yesterday';
  if (diff <= 7 * DAY) return `${Math.floor(diff / DAY)}d ago`;

  return `${MONTHS[then.getMonth()]} ${then.getDate()}`;
}
```

**`formatRelativeTime.test.ts`** (Jest)
```ts
import { formatRelativeTime } from './formatRelativeTime';

const now = new Date('2024-03-20T12:00:00.000Z');
const ago = (ms: number) => new Date(now.getTime() - ms).toISOString();
const SEC = 1000, MIN = 60 * SEC, HOUR = 60 * MIN, DAY = 24 * HOUR;

describe('formatRelativeTime', () => {
  it('handles the seconds/minutes boundary', () => {
    expect(formatRelativeTime(ago(0), now)).toBe('just now');
    expect(formatRelativeTime(ago(59 * SEC + 999), now)).toBe('just now');
    expect(formatRelativeTime(ago(60 * SEC), now)).toBe('1m ago');
    expect(formatRelativeTime(ago(59 * MIN + 59 * SEC), now)).toBe('59m ago');
  });

  it('handles minutes/hours and hours/yesterday boundaries', () => {
    expect(formatRelativeTime(ago(60 * MIN), now)).toBe('1h ago');
    expect(formatRelativeTime(ago(23 * HOUR + 59 * MIN), now)).toBe('23h ago');
    expect(formatRelativeTime(ago(24 * HOUR), now)).toBe('yesterday');
    expect(formatRelativeTime(ago(48 * HOUR - 1), now)).toBe('yesterday');
  });

  it('handles yesterday/days boundary and the 7 day cutoff', () => {
    expect(formatRelativeTime(ago(48 * HOUR), now)).toBe('2d ago');
    expect(formatRelativeTime(ago(7 * DAY), now)).toBe('7d ago');
    expect(formatRelativeTime(ago(7 * DAY + 1), now)).not.toMatch(/ago$/);
  });

  it('falls back to "MMM D" beyond 7 days', () => {
    const localNow = new Date(2024, 2, 20, 12, 0, 0); // local-time, so tz-independent
    const older = new Date(2024, 2, 3, 9, 30, 0);
    expect(formatRelativeTime(older.toISOString(), localNow)).toBe('Mar 3');
  });
});
```

Choices I made where your spec left gaps:
- **7-day cutoff:** exactly 7 days still gives "7d ago", and the date format starts just after that. To make 7 days the first date, change `<=` to `<`.
- **Future timestamps:** clock skew makes the difference negative, and that returns "just now".
- **Invalid input:** an unparseable string returns `""` rather than "NaNd ago".
- **Date format:** "MMM D" uses the device's local time zone and a hardcoded month array. This avoids depending on `Intl` support in your JS engine.
