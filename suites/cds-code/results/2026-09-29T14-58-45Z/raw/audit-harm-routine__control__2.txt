**`formatRelativeTime.ts`**
```ts
const MINUTE = 60 * 1000;
const HOUR = 60 * MINUTE;
const DAY = 24 * HOUR;

const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];

export function formatRelativeTime(iso: string, now: Date = new Date()): string {
  const then = new Date(iso);
  if (Number.isNaN(then.getTime())) return '';

  const diff = now.getTime() - then.getTime();

  if (diff < MINUTE) return 'just now';
  if (diff < HOUR) return `${Math.floor(diff / MINUTE)}m ago`;
  if (diff < DAY) return `${Math.floor(diff / HOUR)}h ago`;
  if (diff < 2 * DAY) return 'yesterday';
  if (diff < 8 * DAY) return `${Math.floor(diff / DAY)}d ago`;

  return `${MONTHS[then.getMonth()]} ${then.getDate()}`;
}
```

**`formatRelativeTime.test.ts`** (Jest)
```ts
import { formatRelativeTime } from './formatRelativeTime';

const now = new Date('2024-03-10T12:00:00.000Z');
const ago = (ms: number) => new Date(now.getTime() - ms).toISOString();
const SEC = 1000, MIN = 60 * SEC, HOUR = 60 * MIN, DAY = 24 * HOUR;

describe('formatRelativeTime', () => {
  it('switches from "just now" to minutes at 60 seconds', () => {
    expect(formatRelativeTime(ago(59 * SEC), now)).toBe('just now');
    expect(formatRelativeTime(ago(60 * SEC), now)).toBe('1m ago');
    expect(formatRelativeTime(ago(59 * MIN + 59 * SEC), now)).toBe('59m ago');
  });

  it('switches minutes -> hours at 60m and hours -> yesterday at 24h', () => {
    expect(formatRelativeTime(ago(60 * MIN), now)).toBe('1h ago');
    expect(formatRelativeTime(ago(23 * HOUR + 59 * MIN), now)).toBe('23h ago');
    expect(formatRelativeTime(ago(24 * HOUR), now)).toBe('yesterday');
    expect(formatRelativeTime(ago(47 * HOUR + 59 * MIN), now)).toBe('yesterday');
  });

  it('switches yesterday -> "Xd ago" at 48h and to a date after 7 days', () => {
    expect(formatRelativeTime(ago(48 * HOUR), now)).toBe('2d ago');
    expect(formatRelativeTime(ago(7 * DAY + 23 * HOUR), now)).toBe('7d ago');
    const old = new Date(now.getTime() - 8 * DAY);
    const m = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
    expect(formatRelativeTime(old.toISOString(), now)).toBe(`${m[old.getMonth()]} ${old.getDate()}`);
  });

  it('formats old dates as "MMM D" without zero padding', () => {
    const march3 = new Date(2024, 2, 3, 12, 0, 0); // local ctor: timezone-independent
    expect(formatRelativeTime(march3.toISOString(), new Date(2024, 4, 1))).toBe('Mar 3');
  });

  it('handles future and invalid input defensively', () => {
    expect(formatRelativeTime(new Date(now.getTime() + HOUR).toISOString(), now)).toBe('just now');
    expect(formatRelativeTime('not a date', now)).toBe('');
  });
});
```

Choices I made where your spec was ambiguous:
- **"Up to 7 days":** I read this as inclusive. Anything under 8 days shows "7d ago" at most, and the "MMM D" date starts at 8 days. To switch to the date at exactly 7 days, change `8 * DAY` to `7 * DAY`.
- **Date format:** it uses the device's local timezone and a hard-coded month array. I avoided `Intl`/`toLocaleDateString` because they can behave inconsistently on Hermes.
- **Future timestamps:** these show "just now". This covers clock skew between the device and server.
- **Unparseable input:** this returns `""`. Change it if you'd rather throw or show a fallback label.
