**`src/utils/formatRelativeTime.ts`**
```ts
const SECOND = 1000;
const MINUTE = 60 * SECOND;
const HOUR = 60 * MINUTE;
const DAY = 24 * HOUR;

const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];

export function formatRelativeTime(iso: string, now: Date = new Date()): string {
  const then = new Date(iso);
  if (Number.isNaN(then.getTime())) return '';

  const diff = now.getTime() - then.getTime();

  if (diff < MINUTE) return 'just now';
  if (diff < HOUR) return `${Math.floor(diff / MINUTE)}m ago`;
  if (diff < DAY) return `${Math.floor(diff / HOUR)}h ago`;
  if (diff < 2 * DAY) return 'yesterday';
  if (diff <= 7 * DAY) return `${Math.floor(diff / DAY)}d ago`;

  return `${MONTHS[then.getMonth()]} ${then.getDate()}`;
}
```

```ts
import { formatRelativeTime } from './formatRelativeTime';

const now = new Date(2024, 5, 15, 12, 0, 0);
const ago = (ms: number) => new Date(now.getTime() - ms).toISOString();
const S = 1000, M = 60 * S, H = 60 * M, D = 24 * H;

describe('formatRelativeTime', () => {
  it('just now / minutes boundary', () => {
    expect(formatRelativeTime(ago(0), now)).toBe('just now');
    expect(formatRelativeTime(ago(59 * S), now)).toBe('just now');
    expect(formatRelativeTime(ago(60 * S), now)).toBe('1m ago');
    expect(formatRelativeTime(ago(59 * M + 59 * S), now)).toBe('59m ago');
  });
  it('minutes / hours / yesterday boundaries', () => {
    expect(formatRelativeTime(ago(60 * M), now)).toBe('1h ago');
    expect(formatRelativeTime(ago(24 * H - S), now)).toBe('23h ago');
    expect(formatRelativeTime(ago(24 * H), now)).toBe('yesterday');
    expect(formatRelativeTime(ago(48 * H - S), now)).toBe('yesterday');
  });
  it('days boundary and 7-day cutoff', () => {
    expect(formatRelativeTime(ago(48 * H), now)).toBe('2d ago');
    expect(formatRelativeTime(ago(7 * D), now)).toBe('7d ago');
    expect(formatRelativeTime(ago(7 * D + S), now)).toBe('Jun 8');
  });
  it('MMM D fallback, future and invalid input', () => {
    expect(formatRelativeTime(new Date(2024, 2, 3, 9, 0).toISOString(), now)).toBe('Mar 3');
    expect(formatRelativeTime(ago(-5 * M), now)).toBe('just now');
    expect(formatRelativeTime('not-a-date', now)).toBe('');
  });
});
```

Choices I made where your spec was ambiguous:
- **7-day cutoff:** exactly 7 days gives "7d ago", and anything past that gives "MMM D".
- **Date format:** the "MMM D" date uses the device's local time zone. It uses a hard-coded month list instead of `Intl`, so it doesn't depend on Hermes locale support.
- **Bad input:** a timestamp in the future gives "just now". An unparseable string gives `""`.
- **Test dates:** the tests build dates with the local `Date` constructor, so they pass in any time zone.
