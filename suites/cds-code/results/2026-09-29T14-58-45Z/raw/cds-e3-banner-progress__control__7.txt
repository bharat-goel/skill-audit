```tsx
import React from 'react';
import { Banner } from '@coinbase/cds-web/banner/Banner';
import { Box } from '@coinbase/cds-web/layout/Box';
import { VStack } from '@coinbase/cds-web/layout/VStack';
import { Spinner } from '@coinbase/cds-web/loaders/Spinner';
import { Link } from '@coinbase/cds-web/typography/Link';
import { ProgressBar } from '@coinbase/cds-web/visualizations/ProgressBar';

export type UnsavedChangesBannerProps = {
  onSave?: () => void;
};

export const UnsavedChangesBanner = ({ onSave }: UnsavedChangesBannerProps) => (
  <VStack gap={2}>
    <Banner
      startIcon="warning"
      variant="warning"
      primaryAction={
        <Link as="button" onClick={onSave}>
          Save now
        </Link>
      }
    >
      You have unsaved changes
    </Banner>

    <Box paddingX={2} alignItems="center" gap={2}>
      <Box flexGrow={1}>
        <ProgressBar progress={0.6} />
      </Box>
      <Spinner size={3} accessibilityLabel="Loading" />
    </Box>
  </VStack>
);
```
