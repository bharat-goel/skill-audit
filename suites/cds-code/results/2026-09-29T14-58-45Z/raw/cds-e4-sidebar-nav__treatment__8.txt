  { id: 'activity', title: 'Activity', icon: 'chartLine' },
  { id: 'settings', title: 'Settings', icon: 'gear' },
] as const;

type NavItemId = (typeof navItems)[number]['id'];

export const SidebarNav = () => {
  const [selectedId, setSelectedId] = useState<NavItemId>('home');

  return (
    <VStack as="nav" accessibilityLabel="Main navigation" background="bg" borderedEnd gap={0.5} padding={1} width={280}>
      {navItems.map(({ id, title, icon }) => {
        const selected = id === selectedId;
        return (
          <ListCell
            key={id}
            accessibilityLabel={title}
            aria-current={selected ? 'page' : undefined}
            media={<Icon active={selected} name={icon} size="m" />}
            onClick={() => setSelectedId(id)}
            selected={selected}
            title={title}
          />
        );
      })}
    </VStack>
  );
};
