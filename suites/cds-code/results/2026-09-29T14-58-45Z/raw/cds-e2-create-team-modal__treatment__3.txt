```tsx
import React, { useCallback, useState } from 'react';
import { Button } from '@coinbase/cds-web/buttons';
import { TextInput } from '@coinbase/cds-web/controls';
import { Select } from '@coinbase/cds-web/alpha/select';
import { VStack } from '@coinbase/cds-web/layout';
import { Modal, ModalBody, ModalFooter, ModalHeader } from '@coinbase/cds-web/overlays';

export type TeamSize = 'small' | 'medium' | 'large';
export type CreateTeamValues = { name: string; size: TeamSize };
export type CreateTeamModalProps = {
  visible: boolean;
  onRequestClose: () => void;
  onSave: (values: CreateTeamValues) => void;
};

const teamSizeOptions = [
  { value: 'small', label: 'Small' },
  { value: 'medium', label: 'Medium' },
  { value: 'large', label: 'Large' },
];

export const CreateTeamModal = ({ visible, onRequestClose, onSave }: CreateTeamModalProps) => {
  const [name, setName] = useState('');
  const [size, setSize] = useState<TeamSize>('small');

  const handleClose = useCallback(() => {
    setName('');
    setSize('small');
    onRequestClose();
  }, [onRequestClose]);

  const handleSave = useCallback(() => {
    onSave({ name: name.trim(), size });
    handleClose();
  }, [name, size, onSave, handleClose]);

  return (
    <Modal onRequestClose={handleClose} visible={visible}>
      <ModalHeader title="Create team" />
      <ModalBody>
        <VStack gap={3}>
          <TextInput label="Team name" onChange={(e) => setName(e.target.value)}
            placeholder="Enter a team name" value={name} />
          <Select label="Team size" onChange={(v) => setSize(v as TeamSize)}
            options={teamSizeOptions} value={size} />
        </VStack>
      </ModalBody>
      <ModalFooter
        primaryAction={<Button disabled={!name.trim()} onClick={handleSave} variant="primary">Save</Button>}
        secondaryAction={<Button onClick={handleClose} variant="secondary">Cancel</Button>}
      />
    </Modal>
  );
};
```
