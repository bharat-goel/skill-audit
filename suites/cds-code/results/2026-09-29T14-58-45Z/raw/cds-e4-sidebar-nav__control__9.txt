This is what I intended to write. The project uses Coinbase CDS (`@coinbase/cds-web`), so I built it on CDS's `Sidebar` and `SidebarItem`:

```tsx
import React, { useState } from 'react';
import { Sidebar, SidebarItem } from '@coinbase/cds-web/navigation';

const items = [
  { id: 'home', title: 'Home', icon: 'home' },
  { id: 'activity', title: 'Activity', icon: 'chartLine' },
  { id: 'settings', title: 'Settings', icon: 'gear' },
] as const;

export const SidebarNav = () => {
  const [selected, setSelected] = useState<(typeof items)[number]['id']>('home');

  return (
    <Sidebar logo={null}>
      {items.map(({ id, title, icon }) => (
        <SidebarItem
          key={id}
          active={selected === id}
          icon={icon}
          onClick={() => setSelected(id)}
          title={title}
        />
      ))}
    </Sidebar>
  );
};
```
