```tsx
import { ContentCard, VStack, HStack, Text, Avatar, ListCell, Divider, Icon } from "@coinbase/cds-web";

export type ProfileCardProps = {
  avatarSrc?: string;
  displayName: string;
  email: string;
  onAccountSettings?: () => void;
  onNotifications?: () => void;
  onSignOut?: () => void;
};

export function ProfileCard({
  avatarSrc,
  displayName,
  email,
  onAccountSettings,
  onNotifications,
  onSignOut,
}: ProfileCardProps) {
  const rows = [
    { label: "Account settings", onClick: onAccountSettings },
    { label: "Notifications", onClick: onNotifications },
    { label: "Sign out", onClick: onSignOut },
  ];

  return (
    <ContentCard borderRadius="roundedLarge" padding={0}>
      <HStack gap={2} alignItems="center" padding={3}>
        <Avatar src={avatarSrc} alt={displayName} size="l" />
        <VStack gap={0}>
          <Text font="title3">{displayName}</Text>
          <Text font="body" color="fgMuted">
            {email}
          </Text>
        </VStack>
      </HStack>

      <Divider />

      <VStack gap={0}>
        {rows.map((row, index) => (
          <ListCell
            key={row.label}
            title={row.label}
            interactive
            onClick={row.onClick}
            end={<Icon name="chevronRight" size="s" color="fgMuted" />}
            borderBottomWidth={index < rows.length - 1 ? "primary" : undefined}
          />
        ))}
      </VStack>
    </ContentCard>
  );
}
```

**Usage:**
```tsx
<ProfileCard
  avatarSrc={user.avatarUrl}
  displayName={user.name}
  email={user.email}
  onAccountSettings={() => navigate("/settings/account")}
  onNotifications={() => navigate("/settings/notifications")}
  onSignOut={handleSignOut}
/>
```

**What's solid:**
- No hardcoded colors/spacing — `padding`, `gap`, `borderRadius` all use CDS scale tokens; `color="fgMuted"` for the email uses the semantic token
- `Text` with `font` props instead of raw typography styles
- `ListCell` for the settings rows per the skill's guidance table
