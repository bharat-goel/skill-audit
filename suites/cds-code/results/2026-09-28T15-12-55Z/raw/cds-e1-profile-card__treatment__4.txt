```tsx
import { VStack, HStack, Text, Avatar, ListCell, Icon, Divider } from "@coinbase/cds-web";

export type ProfileCardProps = {
  avatarSrc?: string;
  displayName: string;
  email: string;
  onAccountSettingsClick?: () => void;
  onNotificationsClick?: () => void;
  onSignOutClick?: () => void;
};

export function ProfileCard({
  avatarSrc,
  displayName,
  email,
  onAccountSettingsClick,
  onNotificationsClick,
  onSignOutClick,
}: ProfileCardProps) {
  return (
    <VStack
      borderWidth={1}
      borderColor="line"
      borderRadius={1}
      background="bg"
      overflow="hidden"
    >
      <HStack gap={2} alignItems="center" padding={3}>
        <Avatar src={avatarSrc} alt={displayName} size="l" />
        <VStack gap={0}>
          <Text font="title4">{displayName}</Text>
          <Text font="body" color="fgMuted">
            {email}
          </Text>
        </VStack>
      </HStack>

      <Divider />

      <VStack>
        <ListCell
          title="Account settings"
          onClick={onAccountSettingsClick}
          end={<Icon name="caretRight" size="s" color="fgMuted" />}
        />
        <ListCell
          title="Notifications"
          onClick={onNotificationsClick}
          end={<Icon name="caretRight" size="s" color="fgMuted" />}
        />
        <ListCell
          title="Sign out"
          onClick={onSignOutClick}
          end={<Icon name="caretRight" size="s" color="fgMuted" />}
        />
      </VStack>
    </VStack>
  );
}
```

**What it uses and why:**
- `VStack`/`HStack` — layout, with `borderRadius={1}` for the subtle rounded corners (CDS numeric spacing/radius scale) and `borderColor="line"` for a light container edge instead of a shadow.
- `Avatar` — the user photo, `size="l"`.
- `Text font="title4"` for the name, `Text font="body" color="fgMuted"` for the muted email — no raw colors/pixels.
- `Divider` separates the identity block from the settings rows.
- `ListCell` for each menu row (`title` prop), with a trailing `Icon name="caretRight"` as the right-side arrow.
