Here's the component:

```tsx
import {
  ContentCard,
  ListCell,
  VStack,
  HStack,
  Text,
  Avatar,
  Divider,
} from "@coinbase/cds-web";
import { Icon } from "@coinbase/cds-icons";

export type ProfileCardProps = {
  displayName: string;
  email: string;
  avatarSrc?: string;
  onAccountSettings?: () => void;
  onNotifications?: () => void;
  onSignOut?: () => void;
};

export function ProfileCard({
  displayName,
  email,
  avatarSrc,
  onAccountSettings,
  onNotifications,
  onSignOut,
}: ProfileCardProps) {
  return (
    <VStack gap={2}>
      <ContentCard>
        <HStack gap={2} alignItems="center">
          <Avatar src={avatarSrc} alt={displayName} size="l" />
          <VStack gap={0}>
            <Text font="title4">{displayName}</Text>
            <Text font="body" color="fgMuted">
              {email}
            </Text>
          </VStack>
        </HStack>
      </ContentCard>

      <ContentCard padding={0}>
        <ListCell
          title="Account settings"
          onClick={onAccountSettings}
          accessory={<Icon name="chevronRight" size="s" color="fgMuted" />}
        />
        <Divider />
        <ListCell
          title="Notifications"
          onClick={onNotifications}
          accessory={<Icon name="chevronRight" size="s" color="fgMuted" />}
        />
        <Divider />
        <ListCell
          title="Sign out"
          onClick={onSignOut}
          accessory={<Icon name="chevronRight" size="s" color="fgMuted" />}
        />
      </ContentCard>
    </VStack>
  );
}
```

Design notes:
- **Container / rounded corners**: `ContentCard` provides CDS's standard card radius by default, so no hardcoded `borderRadius` pixel values were needed — satisfies "subtle rounded corners" via the token system rather than a magic number.
- **Muted email**: `color="fgMuted"` (the semantic token), not a raw gray hex.
- **Menu rows**: `ListCell` with `accessory` for the trailing arrow, wrapped in a second `ContentCard` (padding stripped so `ListCell`'s own row padding applies) with `Divider`s between rows.
- No `TextHeadline`/`TextBody` (deprecated v7 components) — using `Text` with `font` tokens throughout, per the skill's guidance.
