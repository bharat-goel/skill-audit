```tsx
import { Avatar } from "@coinbase/cds-web/avatar";
import { Icon } from "@coinbase/cds-web/icons";
import { ListCell } from "@coinbase/cds-web/list";
import { Box, VStack, HStack } from "@coinbase/cds-web/system";
import { Text } from "@coinbase/cds-web/typography";

type ProfileCardProps = {
  avatarSrc?: string;
  displayName: string;
  email: string;
  onAccountSettings?: () => void;
  onNotifications?: () => void;
  onSignOut?: () => void;
};

export function ProfileCard({
  avatarSrc,
  displayName,
  email,
  onAccountSettings,
  onNotifications,
  onSignOut,
}: ProfileCardProps) {
  return (
    <VStack background="bgAlternate" borderRadius="roundedLarge" overflow="hidden">
      <HStack gap={2} alignItems="center" padding={3}>
        <Avatar src={avatarSrc} name={displayName} size="l" />
        <VStack gap={0}>
          <Text font="title4">{displayName}</Text>
          <Text font="body" color="fgMuted">
            {email}
          </Text>
        </VStack>
      </HStack>

      <Box borderTopWidth={1} borderColor="line" />

      <VStack>
        <ListCell
          title="Account settings"
          onClick={onAccountSettings}
          accessory={<Icon name="caretRight" size="s" color="fgMuted" />}
        />
        <ListCell
          title="Notifications"
          onClick={onNotifications}
          accessory={<Icon name="caretRight" size="s" color="fgMuted" />}
        />
        <ListCell
          title="Sign out"
          onClick={onSignOut}
          accessory={<Icon name="caretRight" size="s" color="fgMuted" />}
        />
      </VStack>
    </VStack>
  );
}
```

Usage:

```tsx
<ProfileCard
  avatarSrc={user.avatarUrl}
  displayName={user.name}
  email={user.email}
  onAccountSettings={() => navigate("/settings/account")}
  onNotifications={() => navigate("/settings/notifications")}
  onSignOut={signOut}
/>
```

What I used and why:
- **`VStack` with `background="bgAlternate"` + `borderRadius="roundedLarge"`** for the card container with subtle rounded corners, `overflow="hidden"` so child content respects the radius.
- **`Avatar`** for the photo (falls back to initials from `name` if `avatarSrc` is empty).
- **`Text font="title4"`** for the display name, **`Text font="body" color="fgMuted"`** for the email — muted per your spec, all through style props rather than inline styles.
- **`ListCell`** for each settings row (the CDS primitive for settings/menu rows), with a trailing caret `Icon` as the right-side arrow.
