```jsx
import styles from './ProfileCard.module.css';

function ChevronRightIcon() {
  return (
    <svg
      width="16"
      height="16"
      viewBox="0 0 16 16"
      fill="none"
      aria-hidden="true"
      className={styles.chevron}
    >
      <path
        d="M6 3.5L10.5 8L6 12.5"
        stroke="currentColor"
        strokeWidth="1.5"
        strokeLinecap="round"
        strokeLinejoin="round"
      />
    </svg>
  );
}

function MenuRow({ label, onClick }) {
  return (
    <button type="button" className={styles.menuRow} onClick={onClick}>
      <span>{label}</span>
      <ChevronRightIcon />
    </button>
  );
}

export function ProfileCard({
  avatarUrl,
  displayName,
  email,
  onAccountSettingsClick,
  onNotificationsClick,
  onSignOutClick,
}) {
  return (
    <div className={styles.card}>
      <div className={styles.header}>
        <img
          src={avatarUrl}
          alt={displayName ? `${displayName}'s avatar` : 'User avatar'}
          className={styles.avatar}
        />
        <div className={styles.info}>
          <span className={styles.displayName}>{displayName}</span>
          <span className={styles.email}>{email}</span>
        </div>
      </div>

      <div className={styles.menu}>
        <MenuRow label="Account settings" onClick={onAccountSettingsClick} />
        <MenuRow label="Notifications" onClick={onNotificationsClick} />
        <MenuRow label="Sign out" onClick={onSignOutClick} />
      </div>
    </div>
  );
}
```
