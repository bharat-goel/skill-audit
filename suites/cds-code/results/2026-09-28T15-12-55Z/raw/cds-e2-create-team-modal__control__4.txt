```tsx
import React, { useState } from 'react';
import { Dialog, DialogHeader, DialogContent, DialogFooter } from '@coinbase/cds-web/overlays';
import { TextInput, Select } from '@coinbase/cds-web/inputs';
import { Button } from '@coinbase/cds-web/buttons';
import { VStack } from '@coinbase/cds-web/layout';

export type TeamSize = 'small' | 'medium' | 'large';

const TEAM_SIZE_OPTIONS: Array<{ label: string; value: TeamSize }> = [
  { label: 'Small', value: 'small' },
  { label: 'Medium', value: 'medium' },
  { label: 'Large', value: 'large' },
];

export type CreateTeamModalProps = {
  isOpen: boolean;
  onClose: () => void;
  onSave: (team: { name: string; size: TeamSize }) => void;
};

export function CreateTeamModal({ isOpen, onClose, onSave }: CreateTeamModalProps) {
  const [name, setName] = useState('');
  const [size, setSize] = useState<TeamSize>('small');

  const canSave = name.trim().length > 0;

  const handleClose = () => {
    setName('');
    setSize('small');
    onClose();
  };

  const handleSave = () => {
    if (!canSave) return;
    onSave({ name: name.trim(), size });
    handleClose();
  };

  return (
    <Dialog visible={isOpen} onRequestClose={handleClose}>
      <DialogHeader title="Create team" />
      <DialogContent>
        <VStack gap={4}>
          <TextInput
            label="Team name"
            placeholder="e.g. Platform Engineering"
            value={name}
            onChange={(e) => setName(e.target.value)}
            autoFocus
          />
          <Select
            label="Team size"
            value={size}
            onChange={(value) => setSize(value as TeamSize)}
            options={TEAM_SIZE_OPTIONS}
          />
        </VStack>
      </DialogContent>
      <DialogFooter>
        <Button variant="secondary" onPress={handleClose}>
          Cancel
        </Button>
        <Button variant="primary" onPress={handleSave} disabled={!canSave}>
          Save
        </Button>
      </DialogFooter>
    </Dialog>
  );
}
```
