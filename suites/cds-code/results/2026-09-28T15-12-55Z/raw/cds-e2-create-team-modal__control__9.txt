```tsx
import React, { useState } from 'react';
import { Dialog, DialogHeader, DialogBody, DialogFooter } from '@coinbase/cds-web/overlays';
import { TextInput, Select } from '@coinbase/cds-web/controls';
import { Button } from '@coinbase/cds-web/buttons';
import { VStack } from '@coinbase/cds-web/layout';

export type TeamSize = 'small' | 'medium' | 'large';

const TEAM_SIZE_OPTIONS: { label: string; value: TeamSize }[] = [
  { label: 'Small', value: 'small' },
  { label: 'Medium', value: 'medium' },
  { label: 'Large', value: 'large' },
];

export type CreateTeamModalProps = {
  visible: boolean;
  onClose: () => void;
  onSave: (team: { name: string; size: TeamSize }) => void;
};

export function CreateTeamModal({ visible, onClose, onSave }: CreateTeamModalProps) {
  const [name, setName] = useState('');
  const [size, setSize] = useState<TeamSize>('small');

  const handleClose = () => {
    setName('');
    setSize('small');
    onClose();
  };

  const handleSave = () => {
    onSave({ name: name.trim(), size });
    handleClose();
  };

  return (
    <Dialog visible={visible} onRequestClose={handleClose}>
      <DialogHeader title="Create team" />
      <DialogBody>
        <VStack gap={2}>
          <TextInput
            label="Team name"
            placeholder="Enter team name"
            value={name}
            onChange={(e) => setName(e.target.value)}
          />
          <Select
            label="Team size"
            value={size}
            onChange={(value) => setSize(value as TeamSize)}
            options={TEAM_SIZE_OPTIONS}
          />
        </VStack>
      </DialogBody>
      <DialogFooter>
        <Button variant="secondary" onPress={handleClose}>
          Cancel
        </Button>
        <Button variant="primary" onPress={handleSave} disabled={!name.trim()}>
          Save
        </Button>
      </DialogFooter>
    </Dialog>
  );
}
```
