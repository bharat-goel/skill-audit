```tsx
import {
  Avatar,
  ContentCard,
  ContentCardBody,
  Divider,
  HStack,
  Icon,
  ListCell,
  Text,
  VStack,
} from "@coinbase/cds-web";

export type ProfileCardProps = {
  avatarSrc?: string;
  displayName: string;
  email: string;
  onAccountSettingsClick?: () => void;
  onNotificationsClick?: () => void;
  onSignOutClick?: () => void;
};

export function ProfileCard({
  avatarSrc,
  displayName,
  email,
  onAccountSettingsClick,
  onNotificationsClick,
  onSignOutClick,
}: ProfileCardProps) {
  return (
    <VStack gap={2}>
      <ContentCard>
        <ContentCardBody>
          <HStack gap={2} alignItems="center">
            <Avatar src={avatarSrc} alt={displayName} size="l" />
            <VStack gap={0}>
              <Text font="title4">{displayName}</Text>
              <Text font="body" color="fgMuted">
                {email}
              </Text>
            </VStack>
          </HStack>
        </ContentCardBody>
      </ContentCard>

      <ContentCard>
        <ListCell
          title="Account settings"
          onClick={onAccountSettingsClick}
          accessory={<Icon name="chevronRight" size="s" color="fgMuted" />}
        />
        <Divider />
        <ListCell
          title="Notifications"
          onClick={onNotificationsClick}
          accessory={<Icon name="chevronRight" size="s" color="fgMuted" />}
        />
        <Divider />
        <ListCell
          title="Sign out"
          onClick={onSignOutClick}
          accessory={<Icon name="chevronRight" size="s" color="fgMuted" />}
        />
      </ContentCard>
    </VStack>
  );
}
```

**What it uses and why (per CDS guidelines):**
- `ContentCard` for both the profile block and the menu group — gives subtle rounded corners for free, no hardcoded border-radius.
- `Avatar` + `Text font="title4"` for the name, `Text font="body" color="fgMuted"` for the muted email.
- `ListCell` for each settings row with a trailing chevron `Icon`, separated by `Divider`.
- All spacing (`gap`, layout) uses CDS tokens, not raw pixels.
