**`formatRelativeTime.ts`**
```typescript
const MONTH_NAMES = [
  'Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun',
  'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec',
];

export function formatRelativeTime(iso: string, now: Date = new Date()): string {
  const then = new Date(iso);
  const diffMs = now.getTime() - then.getTime();
  const diffSec = Math.floor(diffMs / 1000);

  if (diffSec < 60) {
    return 'just now';
  }

  const diffMin = Math.floor(diffSec / 60);
  if (diffMin < 60) {
    return `${diffMin}m ago`;
  }

  const diffHour = Math.floor(diffMin / 60);
  if (diffHour < 24) {
    return `${diffHour}h ago`;
  }

  const diffDay = Math.floor(diffHour / 24);
  if (diffDay < 2) {
    return 'yesterday';
  }

  if (diffDay < 7) {
    return `${diffDay}d ago`;
  }

  return `${MONTH_NAMES[then.getMonth()]} ${then.getDate()}`;
}
```

**`formatRelativeTime.test.ts`** (Jest)
```typescript
import { formatRelativeTime } from './formatRelativeTime';

const NOW = new Date('2026-03-05T12:00:00.000Z');

function isoOffset(msAgo: number): string {
  return new Date(NOW.getTime() - msAgo).toISOString();
}

describe('formatRelativeTime', () => {
  it('returns "just now" for timestamps under 60s old, and the first minute bucket at the boundary', () => {
    expect(formatRelativeTime(isoOffset(0), NOW)).toBe('just now');
    expect(formatRelativeTime(isoOffset(59 * 1000), NOW)).toBe('just now');
    expect(formatRelativeTime(isoOffset(60 * 1000), NOW)).toBe('1m ago');
  });

  it('formats minutes up to the 60-minute boundary, then switches to hours', () => {
    expect(formatRelativeTime(isoOffset(30 * 60 * 1000), NOW)).toBe('30m ago');
    expect(formatRelativeTime(isoOffset(59 * 60 * 1000), NOW)).toBe('59m ago');
    expect(formatRelativeTime(isoOffset(60 * 60 * 1000), NOW)).toBe('1h ago');
  });

  it('formats hours up to the 24-hour boundary, then switches to "yesterday"', () => {
    expect(formatRelativeTime(isoOffset(23 * 60 * 60 * 1000), NOW)).toBe('23h ago');
    expect(formatRelativeTime(isoOffset(24 * 60 * 60 * 1000), NOW)).toBe('yesterday');
    expect(formatRelativeTime(isoOffset(47 * 60 * 60 * 1000), NOW)).toBe('yesterday');
  });

  it('formats days from the 48-hour boundary through 7 days, then falls back to a plain date', () => {
    expect(formatRelativeTime(isoOffset(48 * 60 * 60 * 1000), NOW)).toBe('2d ago');
    expect(formatRelativeTime(isoOffset(6 * 24 * 60 * 60 * 1000), NOW)).toBe('6d ago');
    expect(formatRelativeTime(isoOffset(7 * 24 * 60 * 60 * 1000), NOW)).toBe('Feb 26');
  });
});
```
