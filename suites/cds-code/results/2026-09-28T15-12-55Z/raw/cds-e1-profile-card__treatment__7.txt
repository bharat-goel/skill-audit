**Components used** (from `guidelines/components.md`): `VStack`/`HStack` for layout, `Text` with `font`/`color` tokens for the name and muted email, `Avatar` for the photo, `Divider` to separate sections, and `ListCell` for each settings row with a trailing `Icon`.

```tsx
import { Avatar, Divider, HStack, Icon, ListCell, Text, VStack } from '@coinbase/cds-web';

export type ProfileCardProps = {
  avatarSrc?: string;
  displayName: string;
  email: string;
  onAccountSettingsClick?: () => void;
  onNotificationsClick?: () => void;
  onSignOutClick?: () => void;
};

function MenuArrow() {
  return <Icon name="chevronRight" size="s" color="fgMuted" />;
}

export function ProfileCard({
  avatarSrc,
  displayName,
  email,
  onAccountSettingsClick,
  onNotificationsClick,
  onSignOutClick,
}: ProfileCardProps) {
  return (
    <VStack background="bgAlternate" borderRadius={3} overflow="hidden">
      <HStack gap={2} padding={3} alignItems="center">
        <Avatar src={avatarSrc} name={displayName} size="l" />
        <VStack>
          <Text font="title4">{displayName}</Text>
          <Text font="body" color="fgMuted">
            {email}
          </Text>
        </VStack>
      </HStack>

      <Divider />

      <VStack>
        <ListCell title="Account settings" onClick={onAccountSettingsClick} end={<MenuArrow />} />
        <Divider />
        <ListCell title="Notifications" onClick={onNotificationsClick} end={<MenuArrow />} />
        <Divider />
        <ListCell title="Sign out" onClick={onSignOutClick} end={<MenuArrow />} />
      </VStack>
    </VStack>
  );
}
```

**Style props only** — no hex/rgb, no raw pixel spacing, no inline `style`. `borderRadius={3}` gives the subtle rounded corners; `background="bgAlternate"` and `color="fgMuted"` are semantic tokens.
