**`formatRelativeTime.ts`**
```typescript
const MONTH_NAMES = [
  "Jan", "Feb", "Mar", "Apr", "May", "Jun",
  "Jul", "Aug", "Sep", "Oct", "Nov", "Dec",
];

export function formatRelativeTime(iso: string, now: Date = new Date()): string {
  const then = new Date(iso);
  const diffMs = now.getTime() - then.getTime();
  const diffSec = diffMs / 1000;
  const diffMin = diffSec / 60;
  const diffHour = diffMin / 60;
  const diffDay = diffHour / 24;

  if (diffSec < 60) return "just now";
  if (diffMin < 60) return `${Math.floor(diffMin)}m ago`;
  if (diffHour < 24) return `${Math.floor(diffHour)}h ago`;
  if (diffHour < 48) return "yesterday";
  if (diffDay < 7) return `${Math.floor(diffDay)}d ago`;

  return `${MONTH_NAMES[then.getMonth()]} ${then.getDate()}`;
}
```

**`formatRelativeTime.test.ts`** (Jest) — covers the 60s, 60m/24h, and 48h/7d boundaries:
```typescript
import { formatRelativeTime } from "./formatRelativeTime";

describe("formatRelativeTime", () => {
  const now = new Date("2026-09-28T12:00:00.000Z");

  it("returns 'just now' for timestamps under 60 seconds old", () => {
    const iso = new Date(now.getTime() - 59 * 1000).toISOString();
    expect(formatRelativeTime(iso, now)).toBe("just now");
  });

  it("returns minutes-ago just after the 60s boundary, hours-ago just before 60m", () => {
    expect(formatRelativeTime(new Date(now.getTime() - 60 * 1000).toISOString(), now)).toBe("1m ago");
    expect(formatRelativeTime(new Date(now.getTime() - 59 * 60 * 1000).toISOString(), now)).toBe("59m ago");
  });

  it("returns hours-ago at 60m, and 'yesterday' between 24h and 48h", () => {
    expect(formatRelativeTime(new Date(now.getTime() - 60 * 60 * 1000).toISOString(), now)).toBe("1h ago");
    expect(formatRelativeTime(new Date(now.getTime() - 25 * 60 * 60 * 1000).toISOString(), now)).toBe("yesterday");
  });

  it("returns days-ago up to 7 days, then falls back to 'MMM D'", () => {
    expect(formatRelativeTime(new Date(now.getTime() - 3 * 24 * 60 * 60 * 1000).toISOString(), now)).toBe("3d ago");
    expect(formatRelativeTime(new Date(now.getTime() - 8 * 24 * 60 * 60 * 1000).toISOString(), now)).toBe("Sep 20");
  });
});
```

Note: the fallback date is computed from local time (`getMonth`/`getDate`), so near-midnight timestamps could format as an adjacent day depending on the runner's timezone — flag if you need it pinned to UTC instead.
